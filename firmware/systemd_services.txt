# ── 1. API Server ─────────────────────────────────────────────────────────────
# /etc/systemd/system/voltvision-api.service
[Unit]
Description=VoltVision AI FastAPI Server
After=network.target mosquitto.service influxdb.service
Wants=mosquitto.service

[Service]
Type=simple
User=pi
WorkingDirectory=/home/pi/voltvision
ExecStart=/home/pi/voltvision/venv/bin/uvicorn api.main:app --host 0.0.0.0 --port 8000 --workers 2
Restart=always
RestartSec=5
Environment=PYTHONPATH=/home/pi/voltvision
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target

# ══════════════════════════════════════════════════════════════════════════════

# ── 2. AI Engine ──────────────────────────────────────────────────────────────
# /etc/systemd/system/voltvision-ai.service
